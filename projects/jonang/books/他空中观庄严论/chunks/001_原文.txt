༄། །རྒྱལ་བའི་བཀའ་ཚད་མའི་ལུང་དུ་བསྟན་པ།

༄༅། །ༀ་སྭཱ་སྟི། ན་མོ་བུདྡྷ་ཡ། གང་གི་ཡེ་ཤེས་རྡོ་རྗེའི་མེ་ལྕེ་ཡིས། །བདག་ཏུ་ལྟ་བའི་རི་དང་ནགས་བཅོམ་སྟེ། །རྣམ་ཤེས་བདུན་གནས་ས་འོག་འཇིག་རྟེན་སྐྱོང་། །ལྷ་ཡི་ལྷར་གྱུར་རྒྱལ་བའི་དབང་པོར་འདུད། །གང་དེའི་བསྟན་པ་གནས་སྐབས་ཐ་དད་ཚུལ། །ཐམས་ཅད་ཆོས་ཉིད་འདི་ལ་འབབ་ཚུལ་སྟོན། །འོན་ཀྱང་བྱིས་པ་མི་མཐུན་རྩོད་གྱུར་ལ། །ཡང་དག་རིགས་པའི་གདམས་ངག་སྦྱིན་པར་བྱ། །གངས་ཅན་འདི་ན་མུ་སྟེགས་གྲུབ་མཐའ་དང་། །ཉན་ཐོས་ཚུལ་ལུགས་གཙོ་བོར་འཛིན་པ་མེད། །དེ་དེའི་བག་ཆགས་སད་རྣམས་ཐེག་མཆོག་གི །གཞུང་དོན་ལོག་པར་འགྲེལ་འདི་བཟློག་པར་བྱ། །འཕགས་ཡུལ་མཁས་རྣམས་རྒྱལ་བས་གསུང་ཕྱིན་ཆད། །ལུང་དེ་ཚད་མ་ཉིད་དུ་ཁས་ལེན་ཏེ། །རིགས་པ་ཉེས་པ་ཟད་པ་རྫུན་གྱི་ཚིག །མི་གསུང་ཞེས་འདོད་དེ་ཕྱིར་ཉན་ཐོས་པ། །ཐེག་ཆེན་རྒྱལ་བའི་བཀའ་མིན་ཞེས་ཟེར་ཞིང་། །བཀའ་རུ་གྲུབ་ན་དེ་ཡང་ཁས་ལེན་པས། །ཐེག་ཆེན་བཀའ་སྒྲུབ་བསྟན་འཆོས་རྣམས་ལས་མཛད། །བོད་རྣམས་རང་འདོད་དང་མཚུངས་ལུང་འགའ་ཞིག །འདོད་ཀྱི་གཞན་ཀུན་ཐམས་ཅད་དྲང་དོན་ཏེ། །འགྲོ་ལ་ཕན་དོན་ཡིན་པས་ངོ་མཚར་མོད། །ངོ་བོ་དོན་ལ་མི་གནས་རྫུན་དུ་འདོད། །འདི་ལྟར་འདོད་ལ་ཚད་མའི་ལུང་མི་འགྲུབ། །ཉེས་པ་ཀུན་ཟད་རྒྱལ་བ་རྫུན་གསུང་ན། །ཕྱོགས་རེ་སྤངས་པའི་སློབ་པ་འཕགས་རྣམས་དང་། །ཉེས་ཀུན་དང་བཅས་སོ་སྐྱེ་སྨོས་ཅི་དགོས། །རྫུན་དང་བཅས་པ་སྐྱེས་བུ་ཚད་མ་མིན། །དོན་གྱིས་སངས་རྒྱས་ཚད་མ་མིན་འདོད་པའི། །ནང་པའི་གྲུབ་མཐའ་སྨྲ་བ་དེས་ཅི་རུང་། །ཞེས་པ་ནི་རྒྱལ་བའི་བཀའ་ཚད་མའི་ལུང་དུ་བསྟན་པའོ།། །།

༄། །འཁོར་ལོ་རིམ་པ་གསུམ་གྱི་སྟོན་ཚུལ་ལ་ཡང་དཔྱད་པ།

བོད་རྣམས་ཕལ་ཆེར་འཁོར་ལོ་རིམ་གསུམ་ཡང་། །དང་པོར་ཆོས་ཀུན་བདེན་གྲུབ་བསྟན་ཕྱིར་རྫུན། །བར་པ་སྟོང་ཉིད་བསྟན་པས་ངེས་དོན་ཏེ། །ཕྱི་མར་ཡོད་པ་བསྟན་ཕྱིར་དྲང་དོན་ལོ། །སྤྱིར་ནི་དྲང་དོན་ཐམས་ཅད་རྫུན་ཚིག་མིན། །གནས་ལུགས་མཆོག་ལ་འདྲེན་པའི་ལམ་སྟེགས་ལ། །ཀུན་རྫོབ་བདེན་པ་སྟོན་པ་དྲང་དོན་དང་། །དོན་དམ་དངོས་པོའི་གནས་ལུགས་སྟོན་པ་ལ། །ངེས་པའི་དོན་དུ་གསུངས་ཤིང་མཁས་རྣམས་འདོད། །ཆོས་དེ་གསུངས་པའི་དགོས་པ་བསྟན་པ་ཀུན། །དྲང་དོན་ཡིན་པར་འདོད་མོད་དམ་ཆོས་ཀུན། །གདུལ་བྱའི་དོན་ཕྱིར་དྲང་དོན་ཉིད་དུ་འགྱུར། །ཀླུ་སྒྲུབ་དང་ནི་ཐོགས་མེད་སྐུ་མཆེད་ཀྱིས། །འཁོར་ལོ་རྣམ་གསུམ་དགོངས་པ་གཅིག་ཏུ་བསྒྲུབས། །འཁོར་ལོ་དང་པོར་ཀུན་རྫོབ་གསུངས་པ་ཡང་། །ཇི་ལྟར་སྣང་བ་དེ་ལ་དེ་ལྟར་གསུངས། །གནས་ལུགས་དཔྱད་པའི་དོན་ལ་བདེན་གྲུབ་ཏུ། །བསྟན་པ་མེད་པས་དེ་ལ་རྫུན་ཚིག་མེད། །བར་བར་འཁོར་འདས་ཆོས་ཀུན་བཀག་པ་ཡང་། །ཀུན་རྫོབ་ཀུན་ཏེ་བདེ་གཤེགས་སྙིང་པོ་སོགས། །ཡོད་མེད་གང་ཡང་མ་གསུངས་དཔྱད་པ་མེད། །དེ་ཕྱིར་དེ་གཉིས་ཐ་མ་དོན་མི་འགལ། །འོན་ཀྱང་དང་པོར་ཀུན་རྫོབ་གཙོ་བོར་བསྟན། །བར་དུ་ངེས་དོན་ཕྱེད་ཙམ་ཐ་མར་ནི། །ངེས་དོན་དོན་དམ་རྫོགས་པར་བསྟན་པ་ཡིན། །ནད་པའི་སྨན་དང་ཡི་གེ་སློབ་པའི་དཔེའི། །དགོངས་དོན་འདི་ཡིན་གཞན་ལ་འགལ་བ་ཡོད། །དང་པོར་ཆོས་ཀུན་ཚུགས་ཐུབ་ཡིན་ནོ་ཞེས། །སྟོན་ན་ཀ་ཏྱ་ཡ་ནའི་གདམས་ངག་གི །མདོ་སྡེ་སོགས་དང་ནང་འགལ་ཡོད་པར་འགྱུར། །བར་བར་དོན་དམ་དབྱིངས་སོགས་འགོག་ན་ནི། །བྱམས་ཞུས་སོགས་དང་ནང་འགལ་ཡོད་པར་འགྱུར། །ཞེས་པ་ནི་འཁོར་ལོ་རིམ་པ་གསུམ་གྱི་སྟོན་ཚུལ་ལ་ཡང་དཔྱད་པའོ།། །།

༄། །དྲང་ངེས་ལས་བརྩམས་ཏེ་རྣམ་པར་དཔྱད་པ།

གཞན་ཡང་ཆོས་ཉིད་མི་འགྱུར་བའི་ལེའུར་ཇི་སྐད་དུ། རབ་འབྱོར་འདི་ལྟར་ཆོས་ཐམས་ཅད་ནི་སྤྲུལ་པ་སྟེ། དེ་ལ་ལ་ལ་ནི་ཉན་ཐོས་ཀྱི་སྤྲུལ་པའོ། །ལ་ལ་ནི་རང་སངས་རྒྱས་ཀྱི་སྤྲུལ་པའོ། །ལ་ལ་ནི་བྱང་ཆུབ་སེམས་དཔའི་སྤྲུལ་པའོ། །ལ་ལ་ནི་དེ་བཞིན་གཤེགས་པའི་སྤྲུལ་པའོ། །ལ་ལ་ནི་ཉོན་མོངས་པའི་སྤྲུལ་པའོ། །ལ་ལ་ནི་ཆོས་ཀྱི་སྤྲུལ་པའོ། །རབ་འབྱོར་རྣམ་གྲངས་དེས་ན་ཆོས་ཐམས་ཅད་ནི་སྤྲུལ་པ་ལྟ་བུ་ཡིན་ནོ། །ཞེས་པ་དང་། གསོལ་པ། གང་མ་སྤྲུལ་པའི་ཆོས་དེ་གང་ལགས། བཀའ་སྩལ་པ། རབ་འབྱོར་གང་ལ་སྐྱེ་བ་མེད་པ་དང་། འགག་པ་མེད་པ་དེ་ནི་མ་སྤྲུལ་པའོ། །གསོལ་བ། བཅོམ་ལྡན་འདས་དེ་གང་ལགས། རབ་འབྱོར་དེ་ནི་མི་བསླུ་བའི་ཆོས་ཅན་མྱ་ངན་ལས་འདས་པ་སྟེ། ཆོས་དེ་ནི་མ་སྤྲུལ་པ་ཡིན་ནོ། །ཞེས་མདོ་གསུམ་དུ་མཐུན་པར་གསུངས་སོ། །འདིས་ནི་བདེན་གཉིས་ཡོད་མཉམ་དང་མེད་མཉམ་དུ་འདོད་པ་བསལ་བའོ། །འོ་ན་སྟོང་པ་བདེ་བར་གཤེགས་པས་བསྟན་པ་ལྟར། ངེས་དོན་མདོ་སྡེ་དག་གི་བྱེ་བྲག་ཤེས། །གང་ལས་སེམས་ཅན་གང་ཟག་སྐྱེས་བུར་བསྟན། །ཆོས་དེ་ཐམས་ཅད་དྲང་བའི་དོན་དུ་ཤེས། །ཞེས་པ་འདི་མི་འདོད་དམ་ཞེ་ན། ངེས་པར་འདོད་དོ། །འཁོར་ལོ་ཐ་མར་སྙིང་པོ་བསྟན་པ་ཡང་། །སྟོང་པར་སྟོན་གྱི་སྤྲོས་པར་མི་འཆད་ལ། །དམ་པའི་བདག་སོགས་གང་ཟག་ཏུ་མི་སྟོན། །འདི་དང་བསྟུན་ནའང་ཐ་མ་མཆོག་ཏུ་ཤེས། །དབང་ཕྱུག་སོགས་ཀྱང་འཕགས་པར་བཤད་མོད་ཀྱི། །སྟོན་པར་མི་རུང་དེ་ཚིག་ཚད་མི་བྱ། །ཅི་ཕྱིར་ཞེ་ན་དཀོན་མཆོག་ལས་གཞན་དུ། །བསྟན་ཕྱིར་ཐར་འདོད་རྣམས་ཀྱིས་དོར་བྱ་ཡིན། །འདི་ན་རྒྱལ་གསུང་ཚད་མ་མིན་པ་དང་། །སློབ་དཔོན་འགའ་ཞིག་ཚད་མར་འདོད་པ་ཡོད། །དེ་དག་མུ་སྟེགས་རྣམས་དང་བློ་གྲོས་མཚུངས། །འཕགས་པ་ཡིན་པས་སྐྱོན་དེ་མེད་སྙམ་ན། །དབང་ཕྱུག་སོགས་ལའང་རིགས་པ་དེ་མཚུངས་སོ། །སངས་རྒྱས་ལས་ལྷག་འདོད་ཕྱིར་ནང་པ་མིན། །དེ་ཚིག་རྒྱལ་བའི་གསུང་ལས་ལྷག་པའི་ཕྱིར། །སྐྱེས་བུའང་དེ་ཉིད་མཆོག་ཏུ་ཅིས་མི་འགྱུར། །དེ་དག་རྟོན་པ་བཞི་ཡང་ལོག་པར་སྨྲ། །ཀུན་རྫོབ་ཏུ་ནི་ཡོད་པས་ཡོད་པར་འདོད། །ཀུན་རྫོབ་ཏུ་ནི་མེད་ནའང་མེད་པར་འདོད། །དོན་དམ་དུ་ནི་མེད་ཀྱང་ཡོད་པར་འདོད། །དོན་དམ་མི་རྟོན་ཀུན་རྫོབ་ལ་རྟོན་ནོ། །རྣམ་ཤེས་ངོ་བོར་ཡོད་ན་ཡོད་པར་འདོད། །རྣམ་ཤེས་ངོ་བོར་མེད་ན་མེད་གོ་གཅོད། །ཡེ་ཤེས་སྣང་ངོར་མེད་ཀྱང་ཡོད་ཅེས་ཟེར། །ཡེ་ཤེས་སྤངས་ནས་རྣམ་ཤེས་ཙམ་ལ་རྟོན། །རྒྱལ་བས་གསུངས་པའི་ཆོས་རྣམས་ཕལ་ཆེར་རྫུན། །དགོངས་འགྲེལ་ཚད་ལྡན་ཐམས་ཅད་བདེན་ནོ་ལོ། །རྩ་བ་རྫུན་ན་དགོངས་འབྲེལ་བདེན་མི་སྲིད། །ཆོས་ནི་སྤོང་ཞིང་གང་ཟག་རྣམས་ལ་རྟོན། །དགག་པ་གཉིས་ཀྱི་རྣལ་མར་གོ་བ་སོགས། །རྟོག་གེ་བ་བཞིན་ཚིག་ལ་རྟོན་པའང་ཡོད། །རང་སྟོང་སྨྲ་བ་ཕྱིས་བྱུང་ཕལ་ཆེར་ནི། །རྟོན་པ་བཞི་ལས་རིང་དུ་བྱོལ་བ་ཡིན། །རྒྱལ་བའི་བཀའ་ལས་རྟོན་པ་བཞི་པོ་བསྔགས། །ཞེས་པ་ནི་དྲང་ངེས་ལས་བརྩམས་ཏེ་རྣམ་པར་དཔྱད་པའོ།། །།
